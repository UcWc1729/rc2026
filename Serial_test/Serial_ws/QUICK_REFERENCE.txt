1. cd ~/2026ROBOCON/Task/Serial_test/Serial_ws
2. colcon build && source install/setup.bash
3. sudo chmod 666 /dev/ttyUSB0
4. ros2 launch my_package_py motor_control.launch.py  # 终端1
5. ros2 run my_package_py keyboard_control            # 终端2

┌─────────────────────────────────────────────────────────────────────┐
│ 📡 通信协议 (7字节)                                                  │
└─────────────────────────────────────────────────────────────────────┘
┏━━━━━━━┯━━━━━━━━━┯━━━━━━━━┯━━━━━━┯━━━━━━━┯━━━━━━━┯━━━━━━━┓
┃ 字节  │  0-1    │   2    │  3   │  4-5  │       │   6   ┃
┣━━━━━━━┿━━━━━━━━━┿━━━━━━━━┿━━━━━━┿━━━━━━━┿━━━━━━━┿━━━━━━━┫
┃ 内容  │  帧头   │ 电机   │ 档位 │ CRC16 │       │ 帧尾  ┃
┃       │ 0xAA55  │ 01/02  │ 0-3  │ 校验  │       │ 0x0D  ┃
┗━━━━━━━┷━━━━━━━━━┷━━━━━━━━┷━━━━━━┷━━━━━━━┷━━━━━━━┷━━━━━━━┛

电机类型: 0x01=M3508  0x02=M2006
档  位: 0x00=停止  0x01=低速  0x02=中速  0x03=高速

┌─────────────────────────────────────────────────────────────────────┐
│ 🧪 测试命令                                                          │
└─────────────────────────────────────────────────────────────────────┘
# 查看节点
ros2 node list

# 监控话题
ros2 topic echo /cmd_vel

# 手动发送 (3508低速)
ros2 topic pub --once /cmd_vel geometry_msgs/msg/Twist \
  "{linear: {x: 0.3}, angular: {z: 0.0}}"

# 手动发送 (2006中速)
ros2 topic pub --once /cmd_vel geometry_msgs/msg/Twist \
  "{linear: {x: 0.0}, angular: {z: 0.6}}"

# 协议测试
python3 test_protocol.py
